import { getCollection } from "astro:content";
import PostCard from "@components/PostCard.astro";

export const all = await getCollection("posts");

export const sorted = [...all].sort(
  (a, b) => b.data.date.valueOf() - a.data.date.valueOf()
);

# News
Here you’ll find the latest news and updates from **Immunenotes**. While the website is fully available in English, some news items may appear in German when they relate to German-language sources, topics, or linked content.


{sorted.map((post) => (
  <PostCard
    title={post.data.title}
    date={post.data.date}
    slug={post.id}
    image={post.data.image}
    description={post.data.description}
    sourceType={post.data.sourceType}
    source={post.data.source}
  />
))}
