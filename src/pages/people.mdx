import { getCollection, render } from "astro:content";
import Person from "@components/Person.astro";
import { publications } from "@/data/publications";
import { personPublications } from "@/utils/authorMatch";

export const all = await getCollection("people");

export const firstNameKey = (p) => p.data.name.toLowerCase();

export const sorted = [...all].filter((p) => !p.data.hidden).sort((a, b) =>
  a.data.featured !== b.data.featured
    ? (a.data.featured ? -1 : 1)
    : firstNameKey(a).localeCompare(firstNameKey(b), "de")
);

export const entries = await Promise.all(
  sorted.map(async (entry) => {
    const { Content } = await render(entry);
    return { data: entry.data, Content, publications: personPublications(publications, entry.data) };
  })
);

# People

The **Exercise Immunology and Health Lab** brings together researchers from sports medicine,
immunology, data science, and clinical practice.

{entries.map(({ data, Content, publications }) => (
  <Person
    name={data.name}
    role={data.role}
    mail={data.mail}
    image={data.image}
    alt={data.alt}
    bio={data.bio}
    teaching={data.teaching}
    publications={publications}
    profiles={data.profiles}>
    <Content />
  </Person>
))}
