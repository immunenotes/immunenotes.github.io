import { getCollection } from "astro:content";

export const dashboardOrder = (d) =>
  d.data.order === undefined ? Number.POSITIVE_INFINITY : d.data.order;

export const dashboardDate = (d) =>
  d.data.publishDate ? d.data.publishDate.valueOf() : 0;

export const compareDashboards = (a, b) =>
  dashboardOrder(a) !== dashboardOrder(b)
    ? dashboardOrder(a) - dashboardOrder(b)
    : dashboardDate(a) !== dashboardDate(b)
    ? dashboardDate(b) - dashboardDate(a)
    : a.data.title.localeCompare(b.data.title);

export const dashboards = [...(await getCollection("dashboards"))]
  .filter((d) => !d.data.draft)
  .sort(compareDashboards);
